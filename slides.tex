\begin{frame}{Slide 2 — Challenges We Faced}
\framesubtitle{Technical blockers · Growth problems · Financial pressure}

\begin{columns}[T]

%% LEFT — technical & product challenges
\column{0.5\textwidth}
  \textcolor{rose}{\textbf{\faExclamationTriangle\ Technical \& Product}}\\[5pt]

  \begin{tabular}{@{}p{0.08\linewidth}p{0.82\linewidth}@{}}
    \textcolor{rose}{\faCircle} &
      \scriptsize \textbf{Render → Contabo migration mid-sprint.}
      Render RAM limits caused backend crashes under load.
      Had to migrate VPS and re-configure while continuing to ship features.\\[5pt]

    \textcolor{amber}{\faCircle} &
      \scriptsize \textbf{Guest mode broke the entire auth interceptor.}
      401 errors during guest sessions were triggering the session-expired
      redirect, crashing the app for all unauthenticated visitors.\\[5pt]

    \textcolor{amber}{\faCircle} &
      \scriptsize \textbf{Google Sign-In was silently failing in all browsers.}
      The \texttt{prompt()} One Tap method is suppressed by browser policies.
      Took diagnosis to identify — switched to \texttt{renderButton()}.\\[5pt]

    \textcolor{slate}{\faCircle} &
      \scriptsize \textbf{Database had placeholder/spam assessment content.}
      Microeconomics papers had 1-question placeholders with nonsense titles
      (``examps''). Discovered during a review — manually corrected.\\[5pt]
  \end{tabular}

%% RIGHT — growth & financial challenges
\column{0.46\textwidth}
  \textcolor{rose}{\textbf{\faChartLine\ Growth \& Financial}}\\[5pt]

  \begin{tabular}{@{}p{0.08\linewidth}p{0.82\linewidth}@{}}
    \textcolor{rose}{\faCircle} &
      \scriptsize \textbf{67\% of users sign up and never return.}
      Users were being asked to create accounts before experiencing any value.
      This is the core acquisition bottleneck. Addressed with guest trial.\\[5pt]

    \textcolor{rose}{\faCircle} &
      \scriptsize \textbf{Critical cash position.}
      After next month's committed costs (KES 3,002), only \textbf{KES 144}
      remains. Zero buffer. Revenue or capital injection needed urgently.\\[5pt]

    \textcolor{amber}{\faCircle} &
      \scriptsize \textbf{Zero paying users despite billing being live.}
      The monetisation funnel exists but has not been actively pushed.
      No prompt to upgrade after sign-up.\\[5pt]

    \textcolor{slate}{\faCircle} &
      \scriptsize \textbf{No institutional anchor yet.}
      African Olympiad MOU and IMO team onboarding are pending.
      Without a credible institutional partner, school outreach is harder.\\[5pt]
  \end{tabular}

\end{columns}
\end{frame}
